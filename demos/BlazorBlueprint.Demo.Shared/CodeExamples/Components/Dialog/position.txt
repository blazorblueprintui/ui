<BbDialog Position="DialogPosition.CenterRight">
    <BbDialogTrigger AsChild="false">Open on the Right</BbDialogTrigger>
    <BbDialogContent>
        <BbDialogHeader>
            <BbDialogTitle>User Details</BbDialogTitle>
            <BbDialogDescription>
                This dialog sits on the right side of the viewport.
            </BbDialogDescription>
        </BbDialogHeader>
    </BbDialogContent>
</BbDialog>

@inject DialogService DialogService

private async Task OpenOnTheRight()
{
    await DialogService.OpenAsync<UserDetailsComponent>(
        new DialogOpenOptions
        {
            Title = "User Details",
            Position = DialogPosition.CenterRight
        });
}
