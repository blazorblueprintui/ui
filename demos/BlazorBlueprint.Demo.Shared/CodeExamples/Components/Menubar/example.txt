<BbMenubarItem Shortcut="Ctrl+Shift+T" OnClick="@(() => HandleAction("New Tab"))">
    New Tab
    <BbMenubarShortcut>Ctrl+Shift+T</BbMenubarShortcut>
</BbMenubarItem>

<BbMenubarItem Shortcut="Ctrl+Shift+P" AllowShortcutInEditable="true" OnClick="@(() => HandleAction("Print"))">
    Print...
    <BbMenubarShortcut>Ctrl+Shift+P</BbMenubarShortcut>
</BbMenubarItem>

@* The Shortcut parameter auto-registers the keyboard shortcut globally. *@
@* No need to manually inject IKeyboardShortcutService or manage disposal. *@
