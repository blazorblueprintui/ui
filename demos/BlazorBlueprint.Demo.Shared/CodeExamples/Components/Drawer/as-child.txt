@* Default: the trigger renders its own <button>, so plain content is
   reachable by keyboard and announced as a control. *@
<BbDrawerTrigger>Open the drawer</BbDrawerTrigger>

@* AsChild: the child is already a control, so the trigger renders no button
   of its own. Without this you get a <button> inside a <button>. *@
<BbDrawerTrigger AsChild="true">
    <BbButton Variant="ButtonVariant.Outline">Open the drawer</BbButton>
</BbDrawerTrigger>

@* Same on the close side. *@
<BbDrawerClose>Close</BbDrawerClose>

<BbDrawerClose AsChild="true">
    <BbButton Variant="ButtonVariant.Outline">Close</BbButton>
</BbDrawerClose>
