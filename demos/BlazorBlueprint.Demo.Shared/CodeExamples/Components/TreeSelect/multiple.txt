<BbTreeSelect TItem="Category" Items="categories"
              ValueField="@(c => c.Id)" TextField="@(c => c.Name)"
              ChildrenProperty="@(c => c.Children)"
              Multiple="true" LeafOnly="true" @bind-Values="selectedTeams"
              AriaLabel="Teams" />

@code {
    // Checking Engineering selects runtime, tools and web.
    // With LeafOnly, parent keys are not included in selectedTeams.
    private HashSet<string> selectedTeams = ["runtime", "brand"];
    private readonly List<Category> categories =
    [
        new("engineering", "Engineering", [new("platform", "Platform", [new("runtime", "Runtime", []), new("tools", "Tools", [])]), new("web", "Web", [])]),
        new("design", "Design", [new("product", "Product", []), new("brand", "Brand", [])])
    ];
    private sealed record Category(string Id, string Name, List<Category> Children);
}
