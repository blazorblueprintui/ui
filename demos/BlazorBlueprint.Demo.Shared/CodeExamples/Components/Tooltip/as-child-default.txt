@* v4 default: AsChild is false. The trigger wraps the content in a span that
   carries the hover and focus handlers, so a bare icon just works. *@
<BbTooltip>
    <BbTooltipTrigger>
        <LucideIcon Name="house" Size="18" />
    </BbTooltipTrigger>
    <BbTooltipContent>Home</BbTooltipContent>
</BbTooltip>

@* Set AsChild="true" when the child consumes the trigger context itself,
   such as a BbButton. No wrapper span is rendered. *@
<BbTooltip>
    <BbTooltipTrigger AsChild="true">
        <BbButton Variant="ButtonVariant.Outline">Hover me</BbButton>
    </BbTooltipTrigger>
    <BbTooltipContent>Tooltip on a button</BbTooltipContent>
</BbTooltip>
