@using System.ComponentModel.DataAnnotations
@using BlazorBlueprint.Primitives.Table
@using BlazorBlueprint.Primitives

<BbDataGrid Items="batchRows" TData="Employee" State="batchState" ItemKey="@(r => r.Id)" EditMode="DataGridEditMode.Batch" EditItemFactory="Clone" OnBatchCommit="SaveBatchAsync" ShowPagination="false" AriaLabel="Batch editing employees">
    <Columns>
        <BbDataGridPropertyColumn TData="Employee" TProp="string" Property="@(r => r.Name)" Sortable>
            <EditTemplate><BbInputGroupInput @bind-Value="context.Name" AriaLabel="Name" /></EditTemplate>
        </BbDataGridPropertyColumn>
        <BbDataGridPropertyColumn TData="Employee" TProp="int" Property="@(r => r.Hours)" Sortable>
            <EditTemplate><BbNumericInput TValue="int" @bind-Value="context.Hours" AriaLabel="Hours" ShowButtons="false" /></EditTemplate>
        </BbDataGridPropertyColumn>
    </Columns>
</BbDataGrid>
<p class="text-sm text-muted-foreground" role="status">@batchStatus</p>

@code {
    private readonly List<Employee> batchRows =
    [
        new() { Id = 1, Name = "Alex", Hours = 32 },
        new() { Id = 2, Name = "Sam", Hours = 24 },
        new() { Id = 3, Name = "Morgan", Hours = 40 },
        new() { Id = 4, Name = "Jamie", Hours = 16 }
    ];
    private readonly BlazorBlueprint.Primitives.DataGrid.DataGridState<Employee> batchState = new();
    protected override void OnInitialized() => batchState.Sorting.SetSort("hours", SortDirection.Descending);
    private string batchStatus = "No batch saved yet.";
    private async Task SaveBatchAsync(DataGridBatchCommitContext<Employee> context)
    {
        await Task.Delay(300);
        batchStatus = $"Saved changes to {context.Changes.Count} records.";
    }
    private static Employee Clone(Employee source) => new() { Id = source.Id, Name = source.Name, Hours = source.Hours };
    private sealed class Employee
    {
        public int Id { get; set; }
        [Required] public string Name { get; set; } = "";
        [Range(0, 80)] public int Hours { get; set; }
    }
}
