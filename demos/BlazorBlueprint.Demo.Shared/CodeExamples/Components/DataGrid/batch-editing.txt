@using System.ComponentModel.DataAnnotations
@using BlazorBlueprint.Primitives

        <BbDataGrid Items="batchRows" TData="Employee" ItemKey="@(r => r.Id)" EditMode="DataGridEditMode.Batch" EditItemFactory="Clone" OnBatchCommit="SaveBatchAsync">
            <Columns>
                <BbDataGridPropertyColumn TData="Employee" TProp="string" Property="@(r => r.Name)">
                    <EditTemplate><BbInput @bind-Value="context.Name" /></EditTemplate>
                </BbDataGridPropertyColumn>
                <BbDataGridPropertyColumn TData="Employee" TProp="int" Property="@(r => r.Hours)">
                    <EditTemplate><BbNumericInput TValue="int" @bind-Value="context.Hours" /></EditTemplate>
                </BbDataGridPropertyColumn>
            </Columns>
        </BbDataGrid>

@code {
    private bool rejectSaves;
    private readonly List<Employee> cellRows = [new() { Id = 1, Name = "Alex", Hours = 32 }, new() { Id = 2, Name = "Sam", Hours = 24 }];
    private readonly List<Employee> batchRows = [new() { Id = 1, Name = "Alex", Hours = 32 }, new() { Id = 2, Name = "Sam", Hours = 24 }];
    private static Employee Clone(Employee source) => new() { Id = source.Id, Name = source.Name, Hours = source.Hours };
    private async Task SaveCellAsync(DataGridRowCommitContext<Employee> context)
    {
        await Task.Delay(300);
        context.Cancel = rejectSaves;
    }
    private async Task SaveBatchAsync(DataGridBatchCommitContext<Employee> context)
    {
        await Task.Delay(300);
        context.Cancel = rejectSaves;
    }
    private sealed class Employee
    {
        public int Id { get; set; }
        [Required] public string Name { get; set; } = "";
        [Range(0, 80)] public int Hours { get; set; }
    }
}
